"""
Purpose: Business logic layer for creating and initializing session records.
Flow:
      1. Verify host storage capacity against inbound file size.
      2. Stream upload binary chunks safely onto persistent disk.
      3. Save initial database record using SessionRepository.
Inputs: UploadFile payload, title string, and active AsyncSession.
Outputs: Created SessionModel instance or raises HTTP error exceptions.
"""

import os
import uuid
import pathlib
import aiofiles
from fastapi import UploadFile, HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.config import settings
from app.models.session import SessionModel, SessionStatus
from app.repositories.session_repository import SessionRepository
from app.services.local_storage_service import LocalStorageService

ALLOWED_MIME_TYPES = {
    "audio/mpeg",
    "audio/wav",
    "audio/x-wav",
    "audio/m4a",
    "audio/x-m4a",
    "audio/mp4",
}


class SessionService:
    """
    Purpose: Service handling session creation lifecycle and audio file storage.
    Flow:
          1. Validate audio MIME type.
          2. Check available storage space.
          3. Save audio file locally and record database entry.
    Inputs: Database AsyncSession.
    Outputs: SessionService instance.
    """

    def __init__(self, db: AsyncSession):
        self.repository = SessionRepository(db)

    async def create_session(
        self, title: str, file: UploadFile
    ) -> SessionModel:
        """
        Purpose: Processes uploaded file stream and persists session entity.
        Flow:
              1. Validate file content type.
              2. Inspect disk capacity for required file size.
              3. Stream chunks onto local storage target path.
              4. Commit session record to SQLite database.
        Inputs: title (str), file (UploadFile).
        Outputs: Created SessionModel database object.
        """
        if file.content_type not in ALLOWED_MIME_TYPES:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"Unsupported file type '{file.content_type}'. Allowed types: {', '.join(ALLOWED_MIME_TYPES)}",
            )

        file.file.seek(0, os.SEEK_END)
        file_size = file.file.tell()
        file.file.seek(0)

        LocalStorageService.verify_disk_space(required_bytes=file_size)

        session_id = str(uuid.uuid4())
        file_extension = pathlib.Path(file.filename or "audio.mp3").suffix
        destination_path = (
            pathlib.Path(settings.AUDIO_STORAGE_DIR) / f"{session_id}{file_extension}"
        )

        try:
            async with aiofiles.open(destination_path, "wb") as out_file:
                while chunk := await file.read(1024 * 1024):  # 1MB Chunks
                    await out_file.write(chunk)
        except Exception as err:
            LocalStorageService.cleanup_file(str(destination_path))
            raise HTTPException(
                status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
                detail="Failed to write audio file to local disk.",
            ) from err

        new_session = SessionModel(
            id=session_id,
            title=title,
            file_path=str(destination_path),
            original_file_name=file.filename or "unknown.mp3",
            file_size_bytes=file_size,
            mime_type=file.content_type,
            status=SessionStatus.INGESTION,
        )

        try:
            return await self.repository.create(new_session)
        except Exception:
            LocalStorageService.cleanup_file(str(destination_path))
            raise

    "",
    response_model=SessionResponse,
    status_code=status.HTTP_201_CREATED,
    summary="Upload audio and initiate meeting session",
)
async def create_session(
    background_tasks: BackgroundTasks,
    title: str = Form(..., description="Title of the meeting session"),
    file: UploadFile = File(..., description="Audio recording file upload"),
    db: AsyncSession = Depends(get_db),
):
    """
    Purpose: Endpoint for receiving uploaded audio files and instantiating session record.
    Flow:
          1. Instantiate SessionService context.
          2. Process upload stream and save record to database.
          3. Register background job for asynchronous pipeline execution.
    Inputs: title (Form), file (UploadFile), db session context.
    Outputs: SessionResponse schema model.
    """
    service = SessionService(db)
    session = await service.create_session(title=title, file=file)

    background_tasks.add_task(process_session_background, session.id)

    return session

# TASK-04: Session Ingestion & Upload API Specification

## 1. Overview
TASK-04 implements the primary audio file ingestion endpoint (`POST /api/v1/sessions`). It handles multipart form file uploads, enforces audio MIME type validation, verifies available host disk storage capacity prior to streaming, persists binary payloads onto disk, records metadata in SQLite via `SessionRepository`, and dispatches an asynchronous background task for audio conversion.

---

## 2. API Endpoint Specification

### `POST /api/v1/sessions`
* **Request Content-Type:** `multipart/form-data`
* **Form Fields:**
  * `title` (string, required): Descriptive title for the meeting session.
  * `file` (binary stream, required): Audio recording file (`.mp3`, `.wav`, `.m4a`, `.mp4`).
* **Response Status Codes:**
  * `201 Created`: Upload successful, session created, background pipeline triggered.
  * `400 Bad Request`: File type not permitted or missing required parameters.
  * `507 Insufficient Storage`: Host disk free space is lower than file size.

---

## 3. Storage & Error Cleanup Strategy
* **Chunk Streaming:** Uploads write in 1 MB chunks to prevent RAM exhaustion.
* **Atomic Cleanup:** If database transaction fails after file write, `LocalStorageService.cleanup_file` deletes the orphan audio file.

---

## 4. Operational Flow Architecture

1. Receive multipart form payload -> `create_session()`.
2. Inspect `file.content_type` against `ALLOWED_MIME_TYPES`.
3. Check system free space via `LocalStorageService.verify_disk_space()`.
4. Generate UUID session ID and stream payload to disk location.
5. Create `SessionModel` database record in state `INGESTION`.
6. Attach non-blocking `BackgroundTasks` entry for conversion.
7. Return `SessionResponse` payload with HTTP 201 Created.

"""
Purpose: Validates Task 4 POST /api/v1/sessions upload endpoint behavior.
Flow:
      1. Test successful multipart audio upload stream and 201 response.
      2. Test rejection of unsupported file MIME types (HTTP 400).
Inputs: client (AsyncClient fixture).
Outputs: Pytest assertions.
"""

import io
import pytest
from httpx import AsyncClient


@pytest.mark.asyncio
async def test_create_session_success(client: AsyncClient):
    """
    Purpose: Verify valid audio file upload creates a database session entry.
    Flow:
          1. Create mock audio file byte buffer.
          2. Send POST request with multipart form data to /api/v1/sessions.
          3. Assert response HTTP status 201 Created and correct schema structure.
    """
    fake_audio = io.BytesIO(b"ID3" + b"\x00" * 1000)
    files = {"file": ("test_recording.mp3", fake_audio, "audio/mpeg")}
    data = {"title": "Weekly Engineering Sync"}

    response = await client.post("/api/v1/sessions", data=data, files=files)

    assert response.status_code == 201
    payload = response.json()
    assert payload["title"] == "Weekly Engineering Sync"
    assert payload["status"] == "INGESTION"
    assert payload["original_file_name"] == "test_recording.mp3"


@pytest.mark.asyncio
async def test_create_session_invalid_mime_type(client: AsyncClient):
    """
    Purpose: Verify endpoint rejects non-audio file formats.
    Flow:
          1. Send POST request with text/plain file payload.
          2. Assert HTTP 400 Bad Request error.
    """
    fake_text = io.BytesIO(b"This is a text document, not an audio file.")
    files = {"file": ("notes.txt", fake_text, "text/plain")}
    data = {"title": "Invalid Document"}

    response = await client.post("/api/v1/sessions", data=data, files=files)

    assert response.status_code == 400
    assert "Unsupported file type" in response.json()["detail"]
